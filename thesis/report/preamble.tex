% ---- Page geometry: US Letter, 1 inch margins all sides --------------
\usepackage[letterpaper,margin=1in]{geometry}

% ---- Mathematics (loaded before fontspec, per XeLaTeX guidance) ------
\usepackage{amsmath}
\usepackage{amssymb}

% ---- Fonts: Times New Roman 12pt main text (class option sets 12pt) --
\usepackage{fontspec}
\setmainfont{Times New Roman}
% Monospace falls back to the fontspec default (Latin Modern Mono) so the
% build carries no extra system-font dependency for code listings.

% ---- Languages: English main, Arabic for the mandated AR sections ----
\usepackage{polyglossia}
\setmainlanguage{english}
\setotherlanguage{arabic}
% Times New Roman ships Arabic glyphs, so the AR sections need no extra font.
\newfontfamily\arabicfont[Script=Arabic]{Times New Roman}
% Inline left-to-right run inside RTL text (e.g. Latin figures like "91.4%").
\providecommand{\textLR}[1]{\LR{#1}}

% ---- Line spacing: 1.5 throughout (single-spaced where the template
%      asks: captions, ToC, lists of figures/tables, references) --------
\usepackage{setspace}
\onehalfspacing

% ---- Paragraph shape: template-strict (indented first line, NO extra
%      vertical spacing between paragraphs). -----------------------------
\setlength{\parskip}{0pt}
\setlength{\parindent}{1.8em}

% ---- Line breaking: let long monospace tokens (file paths, Redis channel
%      names, identifiers) move to the next line instead of running into the
%      margin. emergencystretch adds a final relaxed pass; the higher
%      tolerance avoids overfull boxes on lines with unbreakable \texttt runs.
\setlength{\emergencystretch}{3em}
\tolerance=1000
\hyphenpenalty=2000

% ---- Headings: 14pt bold sections, bold numbered subheadings (N.M.K) --
\usepackage{titlesec}
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{3}

  {\normalfont\fontsize{17}{21}\bfseries}{\chaptertitlename\ \thechapter}{1em}{}
\titlespacing*{\chapter}{0pt}{0pt}{20pt}
\titleformat{\section}
  {\normalfont\fontsize{14}{17}\bfseries}{\thesection}{1em}{}
\titleformat{\subsection}
  {\normalfont\fontsize{13}{16}\bfseries}{\thesubsection}{1em}{}
\titleformat{\subsubsection}
  {\normalfont\fontsize{12}{15}\bfseries}{\thesubsubsection}{1em}{}

% =====================================================================
%  BENCHMARK PLACEHOLDERS -- host-migration re-run mechanism.
%  Every benchmark-derived value is a SINGLE named macro defined here and
%  used in the prose/tables. When the new host's results land, edit only
%  the macro body below; the surrounding text, tables, and captions never
%  change. Each placeholder renders visibly as [BENCH:label] and is
%  greppable by the token "BENCH:" or by "\bmtodo".
% =====================================================================
% Base inline marker. The bold is intentional (a deliberately visible
% placeholder) and is exempt from the "no stray bold" formatting rule.
\newcommand{\bmtodo}[1]{\textbf{\textcolor{tbdred}{[BENCH:#1]}}}
% Figure-sized placeholder for a plot whose data is benchmark-derived.
% Renders an upright, fixed-size box at the correct position; never a
% fabricated plot. #1 = short description of the pending figure.
\newcommand{\bmfig}[1]{%
  \begin{center}
  \fcolorbox{slmuted}{slbox}{%
    \parbox[c][55mm][c]{0.85\linewidth}{\centering
      \textbf{\textcolor{tbdred}{[FIGURE PENDING -- updated benchmark run (host migration)]}}\\[4pt]
      \small #1}}%
  \end{center}}
